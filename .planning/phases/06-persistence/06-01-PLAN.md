#pragma once
#include "../defines.hpp"
#include <string>
#include <unordered_map>
#include <functional>

struct SPersistedWindow {
    Vector2D position;
    Vector2D size;
    std::string appClass;
};

struct SPersistedAppDefault {
    Vector2D size;
};

class CCanvasPersistence {
  public:
    void load();
    void scheduleSave();

    std::optional<SPersistedWindow> findByClass(const std::string& appClass) const;
    SPersistedAppDefault getAppDefault(const std::string& appClass) const;
    void trackWindow(const std::string& appClass, const Vector2D& position, const Vector2D& size);
    void removeWindow(const std::string& appClass, const Vector2D& position);

  private:
    void saveNow();
    std::string configPath() const;

    std::vector<SPersistedWindow> m_windows;
    std::unordered_map<std::string, SPersistedAppDefault> m_appDefaults;
    bool m_saveScheduled = false;

    static constexpr int DEBOUNCE_MS = 1000;
};

inline UP<CCanvasPersistence> g_pCanvasPersistence;
```

3. Create src/canvas/CanvasPersistence.cpp:

- `configPath()`: return `$HOME/.config/infinity-land/canvas-state.json`. Use `getenv("HOME")`. Create directory with `std::filesystem::create_directories` if missing.

- `load()`: Read JSON file if it exists. Parse array of window objects `{"class": str, "x": double, "y": double, "w": double, "h": double}` into m_windows. Parse `"app_defaults"` object mapping class -> `{"w": double, "h": double}` into m_appDefaults. Wrap in try/catch — if file missing or malformed, start with empty state (log warning via `Debug::log`).

- `saveNow()`: Serialize m_windows and m_appDefaults to JSON. Write atomically: write to `.tmp` file first, then `std::filesystem::rename` to final path.

- `scheduleSave()`: If m_saveScheduled is true, return. Set m_saveScheduled = true. Use `g_pEventLoopManager->doLater()` to schedule a callback that calls saveNow() after DEBOUNCE_MS. Pattern: `wl_event_loop_add_timer` is complex — instead, use a simpler approach: just call saveNow() directly but guard with m_saveScheduled flag and use `std::chrono` to skip saves within DEBOUNCE_MS of the last save. Store `m_lastSaveTime` as `std::chrono::steady_clock::time_point`. On scheduleSave(): if elapsed < DEBOUNCE_MS, return early. Otherwise saveNow() and update m_lastSaveTime.

- `findByClass()`: Return first matching window from m_windows.

- `getAppDefault()`: Return from m_appDefaults map, or {1200, 800} if not found.

- `trackWindow()`: Update existing entry with same appClass (update position/size), or add new. Also update m_appDefaults[appClass].size = size.

- `removeWindow()`: Remove entry matching appClass + position (approximate match, within 1.0 unit).

Include `<nlohmann/json.hpp>`, `<fstream>`, `<filesystem>`.

void CCanvasLayout::onWindowCreatedTiling(PHLWINDOW pWindow, eDirection) {
    const auto appClass = pWindow->m_initialClass;
    const auto saved = g_pCanvasPersistence->findByClass(appClass);

    Vector2D pos, size;

    if (saved.has_value()) {
        pos  = saved->position;
        size = saved->size;
    } else {
        const auto PMONITOR     = g_pCompositor->getMonitorFromID(pWindow->monitorID());
        const auto screenCenter = PMONITOR->m_position + PMONITOR->m_size / 2.0;
        const auto canvasCenter = g_pCanvasViewport->screenToCanvas(screenCenter);
        const auto appDefault   = g_pCanvasPersistence->getAppDefault(appClass);
        size = appDefault.size;
        pos  = canvasCenter - size / 2.0;
    }

    pWindow->m_position      = pos;
    *pWindow->m_realPosition = pos;
    pWindow->m_size          = size;
    *pWindow->m_realSize     = size;

    m_windows.push_back(pWindow);

    g_pCanvasPersistence->trackWindow(appClass, pos, size);
    g_pCanvasPersistence->scheduleSave();
}
```

4. In `onWindowRemovedTiling()`: Before erasing, save final state and remove from persistence:

void CCanvasLayout::onWindowRemovedTiling(PHLWINDOW pWindow) {
    g_pCanvasPersistence->trackWindow(pWindow->m_initialClass, pWindow->m_position, pWindow->m_size);
    g_pCanvasPersistence->scheduleSave();

}
