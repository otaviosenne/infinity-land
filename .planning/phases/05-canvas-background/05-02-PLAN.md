**1. Update `src/render/shaders/dotgrid.frag.inc`** — add cursor glow uniforms and blending logic:

Add these uniforms:
```glsl
uniform vec2 u_cursorPos;    // cursor in screen pixels (Y-flipped to match gl_FragCoord)
uniform float u_glowRadius;  // glow radius in screen pixels (130.0)
uniform vec4 u_glowDotColor; // brighter dot color for glow
uniform float u_glowDotRadius; // larger radius for glowing dots (2.0)
```

In main(), after computing `dist` to nearest grid dot, add glow logic:

```glsl
// Distance from this grid dot's screen position to cursor
float cursorDist = distance(gridScreenPos, u_cursorPos);

// Glow factor: 1.0 at cursor, 0.0 at glowRadius edge
float glowFactor = 1.0 - smoothstep(0.0, u_glowRadius, cursorDist);

// Interpolate dot properties based on glow
float effectiveRadius = mix(u_dotRadius, u_glowDotRadius, glowFactor);
vec4 effectiveDotColor = mix(u_dotColor, u_glowDotColor, glowFactor);

// Recompute dot alpha with effective radius
float dotAlpha = 1.0 - smoothstep(effectiveRadius - 0.5, effectiveRadius + 0.5, dist);

fragColor = mix(u_bgColor, effectiveDotColor, dotAlpha);
```

```cpp
#pragma once
#include "../defines.hpp"
#include "../helpers/Color.hpp"

enum eCanvasTheme : uint8_t {
    CANVAS_THEME_DARK = 0,
    CANVAS_THEME_LIGHT,
};

struct SCanvasThemeColors {
    CHyprColor bgColor;
    CHyprColor dotColor;
    CHyprColor glowDotColor;
};

class CCanvasTheme {
  public:
    CCanvasTheme();

    void              toggle();
    eCanvasTheme      current() const;
    SCanvasThemeColors colors() const;

  private:
    eCanvasTheme       m_theme = CANVAS_THEME_DARK;

    static const SCanvasThemeColors DARK_COLORS;
    static const SCanvasThemeColors LIGHT_COLORS;
};

inline UP<CCanvasTheme> g_pCanvasTheme;
```

**2. Create `src/canvas/CanvasTheme.cpp`**:

```cpp
#include "CanvasTheme.hpp"
#include "CanvasViewport.hpp"

const SCanvasThemeColors CCanvasTheme::DARK_COLORS = {
    CHyprColor(0.102f, 0.102f, 0.180f, 1.0f),  // bg: #1a1a2e
    CHyprColor(0.216f, 0.255f, 0.318f, 1.0f),  // dot: #374151
    CHyprColor(0.612f, 0.639f, 0.686f, 1.0f),  // glow: #9CA3AF
};

const SCanvasThemeColors CCanvasTheme::LIGHT_COLORS = {
    CHyprColor(0.961f, 0.961f, 0.961f, 1.0f),  // bg: #f5f5f5
    CHyprColor(0.820f, 0.836f, 0.855f, 1.0f),  // dot: #D1D5DB
    CHyprColor(0.612f, 0.639f, 0.686f, 1.0f),  // glow: #9CA3AF
};

CCanvasTheme::CCanvasTheme() = default;

void CCanvasTheme::toggle() {
    m_theme = (m_theme == CANVAS_THEME_DARK) ? CANVAS_THEME_LIGHT : CANVAS_THEME_DARK;
    if (g_pCanvasViewport)
        g_pCanvasViewport->damageAllMonitors();
}

eCanvasTheme CCanvasTheme::current() const {
    return m_theme;
}

SCanvasThemeColors CCanvasTheme::colors() const {
    return (m_theme == CANVAS_THEME_DARK) ? DARK_COLORS : LIGHT_COLORS;
}
```
